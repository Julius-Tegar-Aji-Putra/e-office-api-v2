📊 Dokumentasi API Dashboard - E-Office ST/SK Dekan

🎯 Overview
Dashboard menyediakan tampilan data surat yang berbeda untuk setiap role. Setiap role memiliki kolom tabel, filter, dan aksi yang berbeda sesuai dengan tanggung jawab mereka dalam workflow.

🌐 Base URL
Local: http://localhost:3079
Network: http://192.168.18.36:3079

🔐 Authentication Required
Semua endpoint dashboard memerlukan autentikasi. Pastikan sudah login dan cookie better-auth.session_token terkirim.

Tambahan penting (revisi): setelah login frontend wajib panggil GET /me untuk mendapatkan informasi user lengkap termasuk role.
Example:
GET /me
Response:
{
  "id": "...",
  "name": "Dr. Eko Prasetyo",
  "email": "kaprodi.if@undip.ac.id",
  "role": "KAPRODI",        // primary role
  "roles": ["KAPRODI"],     // semua role jika multi-role
  "profile": { /* ... */ }
}

 // Axios setup
const api = axios.create({
  baseURL: 'http://localhost:3079',
  withCredentials: true // WAJIB!
});

📍 Dashboard Endpoints

1. Get Dashboard (Unified)
Endpoint utama yang mengembalikan data dashboard sesuai role user yang login.

Endpoint: GET /dash

Query Parameters:
{
  page?: number;        // Default: 1
  limit?: number;       // Default: 20
  search?: string;      // Cari berdasarkan judul/nama pengaju
  status?: string;      // Filter by status
  letterType?: string;  // Filter by "SURAT_TUGAS" | "SURAT_KEPUTUSAN"
  category?: string;    // Filter by "AKADEMIK" | "SUMBER_DAYA" | "UMUM"
  type?: string;        // "masuk" | "keluar" (untuk semua role fakultas)
}

Headers:
Cookie: better-auth.session_token=eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...

🎨 Frontend Implementation Examples

React + Axios:

// services/dashboard.service.ts
import axios from 'axios';

const api = axios.create({
  baseURL: 'http://localhost:3079',
  withCredentials: true
});

export const dashboardService = {
  getDashboard: (params?: {
    page?: number;
    limit?: number;
    search?: string;
    status?: string;
    letterType?: string;
    category?: string;
    type?: string;
  }) => api.get('/dash', { params }),
  
  getStatistics: () => api.get('/dash/statistics'),
  
  getRecent: () => api.get('/dash/recent')
};

// components/Dashboard.tsx
import { useEffect, useState } from 'react';
import { dashboardService } from '../services/dashboard.service';

interface DashboardData {
  items: any[];
  pagination: any;
  columns: Array<{ key: string; label: string }>;
  filters: any;
  availableActions?: string[];
  tabs?: Array<{ key: string; label: string; count: number }>;
}

export const Dashboard = () => {
  const [data, setData] = useState<DashboardData | null>(null);
  const [loading, setLoading] = useState(true);
  const [filters, setFilters] = useState({
    page: 1,
    limit: 20,
    search: '',
    status: '',
    type: 'masuk' // Default untuk role fakultas
  });

  useEffect(() => {
    loadDashboard();
  }, [filters]);

  const loadDashboard = async () => {
    try {
      setLoading(true);
      const response = await dashboardService.getDashboard(filters);
      setData(response.data.data);
    } catch (error) {
      console.error('Failed to load dashboard:', error);
    } finally {
      setLoading(false);
    }
  };

  if (loading) return <div>Loading...</div>;
  if (!data) return <div>No data</div>;

  return (
    <div>
      {/* Tab Switcher untuk Role Fakultas */}
      {data.tabs && (
        <div className="tabs">
          {data.tabs.map(tab => (
            <button
              key={tab.key}
              className={filters.type === tab.key ? 'active' : ''}
              onClick={() => setFilters({ ...filters, type: tab.key })}
            >
              {tab.label} ({tab.count})
            </button>
          ))}
        </div>
      )}

      {/* Filters */}
      <div className="filters">
        <input
          type="text"
          placeholder="Search..."
          value={filters.search}
          onChange={(e) => setFilters({ ...filters, search: e.target.value })}
        />
        
        {data.filters.type && (
          <select
            value={filters.type}
            onChange={(e) => setFilters({ ...filters, type: e.target.value })}
          >
            <option value="masuk">Surat Masuk</option>
            <option value="keluar">Surat Keluar</option>
          </select>
        )}
        
        {data.filters.status && (
          <select
            value={filters.status}
            onChange={(e) => setFilters({ ...filters, status: e.target.value })}
          >
            <option value="">All Status</option>
            {data.filters.status.map((status: string) => (
              <option key={status} value={status}>{status}</option>
            ))}
          </select>
        )}
      </div>

      {/* Table */}
      <table>
        <thead>
          <tr>
            {data.columns.map(col => (
              <th key={col.key}>{col.label}</th>
            ))}
          </tr>
        </thead>
        <tbody>
          {data.items.map(item => (
            <tr key={item.id}>
              {data.columns.map(col => (
                <td key={col.key}>
                  {col.key === 'actions' ? (
                    <div>
                      {item.actions.map((action: string) => (
                        <button key={action}>{action}</button>
                      ))}
                    </div>
                  ) : col.key === 'tanggalSurat' ? (
                    new Date(item[col.key]).toLocaleDateString('id-ID')
                  ) : (
                    item[col.key]
                  )}
                </td>
              ))}
            </tr>
          ))}
        </tbody>
      </table>

      {/* Pagination */}
      <div className="pagination">
        <button
          disabled={filters.page === 1}
          onClick={() => setFilters({ ...filters, page: filters.page - 1 })}
        >
          Previous
        </button>
        <span>Page {data.pagination.page} of {data.pagination.totalPages}</span>
        <button
          disabled={filters.page === data.pagination.totalPages}
          onClick={() => setFilters({ ...filters, page: filters.page + 1 })}
        >
          Next
        </button>
      </div>
    </div>
  );
};

404 | Letter not found | ID surat tidak valid
500 | Internal Server Error | Cek server logs

import axios from 'axios';

const api = axios.create({

export const dashboardService = {
  getDashboard: (params?: {
    page?: number;
    limit?: number;
    search?: string;
    status?: string;
    letterType?: string;
    category?: string;

// components/Dashboard.tsx
import { useEffect, useState } from 'react';
import { dashboardService } from '../services/dashboard.service';

interface DashboardData {
  items: any[];
  pagination: any;
  columns: Array<{ key: string; label: string }>;
  filters: any;
  availableActions?: string[];
  tabs?: Array<{ key: string; label: string; count: number }>;
}

export const Dashboard = () => {
  const [data, setData] = useState<DashboardData | null>(null);
  const [loading, setLoading] = useState(true);
  const [filters, setFilters] = useState({
    page: 1,
    limit: 20,
    search: '',
    status: '',
    type: 'masuk' // Default untuk role fakultas
  });

  useEffect(() => {
    loadDashboard();
  }, [filters]);

  const loadDashboard = async () => {
    try {
      setLoading(true);
      const response = await dashboardService.getDashboard(filters);
      setData(response.data.data);
    } catch (error) {
      console.error('Failed to load dashboard:', error);
    } finally {
      setLoading(false);
    }
  };

  if (loading) return <div>Loading...</div>;
  if (!data) return <div>No data</div>;

  return (
    <div>
      {/* Tab Switcher untuk Role Fakultas */}
      {data.tabs && (
        <div className="tabs">
          {data.tabs.map(tab => (
            <button
              key={tab.key}
              className={filters.type === tab.key ? 'active' : ''}
              onClick={() => setFilters({ ...filters, type: tab.key })}
            >
              {tab.label} ({tab.count})
            </button>
          ))}
        </div>
      )}

      {/* Filters */}
      <div className="filters">
        <input
          type="text"
          placeholder="Search..."
          value={filters.search}
          onChange={(e) => setFilters({ ...filters, search: e.target.value })}
        />
        
        {data.filters.type && (
          <select
            value={filters.type}
            onChange={(e) => setFilters({ ...filters, type: e.target.value })}
          >
            <option value="masuk">Surat Masuk</option>
            <option value="keluar">Surat Keluar</option>
          </select>
        )}
        
        {data.filters.status && (
          <select
            value={filters.status}
            onChange={(e) => setFilters({ ...filters, status: e.target.value })}
          >
            <option value="">All Status</option>
            {data.filters.status.map((status: string) => (
              <option key={status} value={status}>{status}</option>
            ))}
          </select>
        )}
      </div>

      {/* Table */}
      <table>
        <thead>
          <tr>
            {data.columns.map(col => (
              <th key={col.key}>{col.label}</th>
            ))}
          </tr>
        </thead>
        <tbody>
          {data.items.map(item => (
            <tr key={item.id}>
              {data.columns.map(col => (
                <td key={col.key}>
                  {col.key === 'actions' ? (
                    <div>
                      {item.actions.map((action: string) => (
                        <button key={action}>{action}</button>
                      ))}
                    </div>
                  ) : col.key === 'tanggalSurat' ? (
                    new Date(item[col.key]).toLocaleDateString('id-ID')
                  ) : (
                    item[col.key]
                  )}
                </td>
              ))}
            </tr>
          ))}
        </tbody>
      </table>

      {/* Pagination */}
      <div className="pagination">
        <button
          disabled={filters.page === 1}
          onClick={() => setFilters({ ...filters, page: filters.page - 1 })}
        >
          Previous
        </button>
        <span>Page {data.pagination.page} of {data.pagination.totalPages}</span>
        <button
          disabled={filters.page === data.pagination.totalPages}
          onClick={() => setFilters({ ...filters, page: filters.page + 1 })}
        >
          Next
        </button>
      </div>
    </div>
  );
};
